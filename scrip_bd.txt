CREATE DATABASE IF NOT EXISTS sistema_tutorias
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE sistema_tutorias;
SET FOREIGN_KEY_CHECKS = 1;

-- ============================================================
--  BLOQUE 1 — CATÁLOGOS MAESTROS Y SEGURIDAD
-- ============================================================

CREATE TABLE carrera (
    id      INT          NOT NULL AUTO_INCREMENT,
    nombre  VARCHAR(120) NOT NULL,
    clave   VARCHAR(20)  NOT NULL,
    estado  ENUM('activa','inactiva') NOT NULL DEFAULT 'activa',
    CONSTRAINT pk_carrera       PRIMARY KEY (id),
    CONSTRAINT uq_carrera_nombre UNIQUE (nombre),
    CONSTRAINT uq_carrera_clave  UNIQUE (clave)
) ENGINE=InnoDB COMMENT='Catálogo de carreras del instituto';

-- NUEVA TABLA: Grupo (Para eliminar dependencia transitiva en Asignación)
CREATE TABLE grupo (
    id          INT          NOT NULL AUTO_INCREMENT,
    nombre      VARCHAR(20)  NOT NULL COMMENT 'Ej: 6-A',
    carrera_id  INT          NOT NULL,
    CONSTRAINT pk_grupo PRIMARY KEY (id),
    CONSTRAINT fk_grupo_carrera FOREIGN KEY (carrera_id) REFERENCES carrera (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Catálogo de grupos por carrera';

CREATE TABLE usuario (
    id                    INT          NOT NULL AUTO_INCREMENT,
    numero_identificacion VARCHAR(30)  NOT NULL COMMENT 'Número de empleado o matrícula',
    nombre_completo       VARCHAR(150) NOT NULL,
    correo                VARCHAR(120) NOT NULL,
    contrasena_hash       VARCHAR(255) NOT NULL,
    foto_perfil           VARCHAR(255) NULL     COMMENT 'Ruta relativa JPG/PNG',
    estado                ENUM('activo','inactivo') NOT NULL DEFAULT 'activo',
    created_at            DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_usuario          PRIMARY KEY (id),
    CONSTRAINT uq_usuario_num_id   UNIQUE (numero_identificacion),
    CONSTRAINT uq_usuario_correo   UNIQUE (correo)
) ENGINE=InnoDB COMMENT='Credenciales de acceso al sistema';

CREATE TABLE perfil (
    id     INT         NOT NULL AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL COMMENT 'ADMINISTRADOR | TUTOR | TUTORADO | COORDINADOR_CARRERA',
    CONSTRAINT pk_perfil       PRIMARY KEY (id),
    CONSTRAINT uq_perfil_nombre UNIQUE (nombre)
) ENGINE=InnoDB COMMENT='Catálogo de roles del sistema';

CREATE TABLE usuario_perfil (
    usuario_id INT NOT NULL,
    perfil_id  INT NOT NULL,
    CONSTRAINT pk_usuario_perfil PRIMARY KEY (usuario_id, perfil_id),
    CONSTRAINT fk_up_usuario
        FOREIGN KEY (usuario_id) REFERENCES usuario (id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_up_perfil
        FOREIGN KEY (perfil_id)  REFERENCES perfil (id)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='Asignación de perfiles a usuarios';

CREATE TABLE tutor (
    id         INT NOT NULL AUTO_INCREMENT,
    usuario_id INT NOT NULL,
    CONSTRAINT pk_tutor          PRIMARY KEY (id),
    CONSTRAINT uq_tutor_usuario  UNIQUE (usuario_id),
    CONSTRAINT fk_tutor_usuario
        FOREIGN KEY (usuario_id) REFERENCES usuario (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Datos específicos del rol Tutor';

CREATE TABLE tutorado (
    id         INT          NOT NULL AUTO_INCREMENT,
    usuario_id INT          NOT NULL,
    carrera_id INT          NOT NULL,
    semestre   TINYINT UNSIGNED NOT NULL DEFAULT 1 
               COMMENT 'Semestre actual del tutorado',
    CONSTRAINT pk_tutorado         PRIMARY KEY (id),
    CONSTRAINT uq_tutorado_usuario UNIQUE (usuario_id),
    CONSTRAINT fk_tutorado_usuario
        FOREIGN KEY (usuario_id) REFERENCES usuario (id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_tutorado_carrera
        FOREIGN KEY (carrera_id) REFERENCES carrera (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Datos específicos del rol Tutorado';

-- NUEVA TABLA: Coordinador de Carrera
CREATE TABLE coordinador_carrera (
    id         INT NOT NULL AUTO_INCREMENT,
    usuario_id INT NOT NULL,
    carrera_id INT NOT NULL,
    CONSTRAINT pk_coordinador_carrera     PRIMARY KEY (id),
    CONSTRAINT uq_coordinador_usuario     UNIQUE (usuario_id),
    CONSTRAINT fk_coordinador_usuario
        FOREIGN KEY (usuario_id) REFERENCES usuario (id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_coordinador_carrera
        FOREIGN KEY (carrera_id) REFERENCES carrera (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Datos específicos del rol Coordinador de Carrera';

CREATE TABLE periodo_semestral (
    id           INT         NOT NULL AUTO_INCREMENT,
    clave        VARCHAR(20) NOT NULL COMMENT 'Ej: ENE-JUN-2025',
    fecha_inicio DATE        NOT NULL,
    fecha_fin    DATE        NOT NULL,
    estado       ENUM('activo','cerrado') NOT NULL DEFAULT 'activo',
    CONSTRAINT pk_periodo        PRIMARY KEY (id),
    CONSTRAINT uq_periodo_clave  UNIQUE (clave),
    CONSTRAINT chk_periodo_fechas CHECK (fecha_fin > fecha_inicio)
) ENGINE=InnoDB COMMENT='Periodos semestrales del programa de tutorías';

-- ============================================================
--  BLOQUE 2 — ASIGNACIONES
-- ============================================================

CREATE TABLE asignacion (
    id            INT         NOT NULL AUTO_INCREMENT,
    tutor_id      INT         NOT NULL COMMENT 'FK a tutor.id',
    grupo_id      INT         NOT NULL COMMENT 'FK a grupo.id',
    periodo_id    INT         NOT NULL,
    aula          VARCHAR(30) NOT NULL,
    dia_horario   VARCHAR(30) NOT NULL DEFAULT 'Viernes',
    hora_horario      TIME        NOT NULL,
    hora_fin_horario  TIME        NOT NULL,
    CONSTRAINT pk_asignacion PRIMARY KEY (id),
    CONSTRAINT uq_asignacion UNIQUE (tutor_id, grupo_id, hora_horario, periodo_id),
    CONSTRAINT fk_asignacion_tutor
        FOREIGN KEY (tutor_id)   REFERENCES tutor (id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_asignacion_grupo
        FOREIGN KEY (grupo_id)   REFERENCES grupo (id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_asignacion_periodo
        FOREIGN KEY (periodo_id) REFERENCES periodo_semestral (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Asignación de tutor a grupo en un periodo';

CREATE TABLE asignacion_tutorado (
    id            INT NOT NULL AUTO_INCREMENT,
    asignacion_id INT NOT NULL,
    tutorado_id   INT NOT NULL COMMENT 'FK a tutorado.id',
    CONSTRAINT pk_asignacion_tutorado PRIMARY KEY (id),
    CONSTRAINT uq_asig_tutorado UNIQUE (asignacion_id, tutorado_id),
    CONSTRAINT fk_asig_tut_asignacion
        FOREIGN KEY (asignacion_id) REFERENCES asignacion (id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_asig_tut_tutorado
        FOREIGN KEY (tutorado_id)   REFERENCES tutorado (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Relación muchos-a-muchos asignacion↔tutorado';

-- ============================================================
--  BLOQUE 3 — SESIONES Y ASISTENCIA
-- ============================================================

CREATE TABLE sesion (
    id            INT          NOT NULL AUTO_INCREMENT,
    asignacion_id INT          NOT NULL,
    fecha         DATE         NOT NULL,
    hora_inicio   TIME         NOT NULL,
    hora_fin      TIME         NULL,
    evidencia     VARCHAR(255) NULL COMMENT 'Ruta relativa JPG/PNG/PDF',
    CONSTRAINT pk_sesion PRIMARY KEY (id),
    CONSTRAINT uq_sesion UNIQUE (asignacion_id, fecha),
    CONSTRAINT chk_sesion_horas CHECK (hora_fin IS NULL OR hora_fin > hora_inicio),
    CONSTRAINT fk_sesion_asignacion
        FOREIGN KEY (asignacion_id) REFERENCES asignacion (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Sesiones de tutoría por asignación';

CREATE TABLE registro_asistencia (
    id                INT  NOT NULL AUTO_INCREMENT,
    sesion_id         INT  NOT NULL,
    tutorado_id       INT  NOT NULL COMMENT 'FK a tutorado.id',
    estado_asistencia ENUM('presente','ausente','justificado') NOT NULL,
    CONSTRAINT pk_registro_asistencia PRIMARY KEY (id),
    CONSTRAINT uq_registro_asistencia UNIQUE (sesion_id, tutorado_id),
    CONSTRAINT fk_reg_asist_sesion
        FOREIGN KEY (sesion_id)   REFERENCES sesion (id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_reg_asist_tutorado
        FOREIGN KEY (tutorado_id) REFERENCES tutorado (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Asistencia de cada tutorado por sesión';

-- ============================================================
--  BLOQUE 4 — PLAN DE ACCIÓN TUTORIAL (PAT) - NUEVO FLUJO
-- ============================================================

-- Actividades PAT generales creadas por el Administrador DDA (para todas las carreras)
CREATE TABLE actividad_pat_general (
    id             INT          NOT NULL AUTO_INCREMENT,
    tipo_actividad ENUM('taller','asesoria_academica',
                        'sesion_informativa','otra') NOT NULL,
    titulo         VARCHAR(200) NOT NULL,
    descripcion    TEXT         NULL,
    fecha_registro DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_actividad_pat_general PRIMARY KEY (id)
) ENGINE=InnoDB COMMENT='Actividades PAT generales creadas por Admin DDA';

-- Actividades PAT adaptadas por el Coordinador de Carrera a su carrera correspondiente
CREATE TABLE actividad_pat_carrera (
    id                    INT          NOT NULL AUTO_INCREMENT,
    actividad_general_id  INT          NOT NULL COMMENT 'FK a actividad_pat_general.id',
    carrera_id            INT          NOT NULL COMMENT 'FK a carrera.id',
    titulo_adaptado       VARCHAR(200) NOT NULL,
    descripcion_adaptada  TEXT         NULL,
    observaciones         TEXT         NULL,
    fecha_adaptacion      DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_actividad_pat_carrera PRIMARY KEY (id),
    CONSTRAINT uq_actividad_carrera UNIQUE (actividad_general_id, carrera_id),
    CONSTRAINT fk_act_carrera_general
        FOREIGN KEY (actividad_general_id) REFERENCES actividad_pat_general (id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_act_carrera_carrera
        FOREIGN KEY (carrera_id) REFERENCES carrera (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Actividades PAT adaptadas por Coordinador a su carrera';

-- Vinculación de actividades adaptadas a sesiones (el tutor las selecciona al crear sesión)
CREATE TABLE sesion_actividad (
    id                    INT          NOT NULL AUTO_INCREMENT,
    sesion_id             INT          NOT NULL COMMENT 'FK a sesion.id',
    actividad_carrera_id  INT          NOT NULL COMMENT 'FK a actividad_pat_carrera.id',
    observaciones         TEXT         NULL,
    evidencia             VARCHAR(255) NULL COMMENT 'Ruta relativa JPG/PNG/PDF',
    fecha_registro        DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_sesion_actividad PRIMARY KEY (id),
    CONSTRAINT uq_sesion_actividad UNIQUE (sesion_id, actividad_carrera_id),
    CONSTRAINT fk_sesact_sesion
        FOREIGN KEY (sesion_id) REFERENCES sesion (id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_sesact_actividad
        FOREIGN KEY (actividad_carrera_id) REFERENCES actividad_pat_carrera (id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB COMMENT='Actividades PAT vinculadas a sesiones por el tutor';

CREATE TABLE plan_sesion_carrera (
    id                    INT  NOT NULL AUTO_INCREMENT,
    carrera_id            INT  NOT NULL COMMENT 'FK a carrera.id',
    numero_sesion         INT  NOT NULL COMMENT 'Número de sesión semanal (1-10)',
    actividad_carrera_id  INT  NOT NULL COMMENT 'FK a actividad_pat_carrera.id',
    CONSTRAINT pk_plan_sesion_carrera PRIMARY KEY (id),
    CONSTRAINT uq_plan_sesion_carrera UNIQUE (carrera_id, numero_sesion),
    CONSTRAINT fk_plan_sesion_carrera
        FOREIGN KEY (carrera_id) REFERENCES carrera (id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_plan_sesion_actividad
        FOREIGN KEY (actividad_carrera_id) REFERENCES actividad_pat_carrera (id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT chk_numero_sesion CHECK (numero_sesion BETWEEN 1 AND 10)
) ENGINE=InnoDB COMMENT='Plan semanal de actividades organizado por el coordinador de carrera';
